pipeline {
  agent any

  parameters {
    booleanParam(name: 'APPLY', defaultValue: false, description: 'Apply Terraform?')
  }

  stages {
    stage('Checkout') {
      steps { checkout scm }
    }

    stage('Init') {
      steps { sh "cd terraform && terraform init -input=false" }
    }

    stage('Validate') {
      steps { sh "cd terraform && terraform validate" }
    }

    stage('Plan') {
      steps { sh "cd terraform && terraform plan -out=tfplan" }
    }

    stage('Apply') {
      when { expression { params.APPLY == true } }
      steps { sh "cd terraform && terraform apply -auto-approve tfplan" }
    }
  }
}
